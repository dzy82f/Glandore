wp db query "
	SELECT
  s.id,
  s.thread_id,
  s.version,
  s.created_at,

  /* header: first <h3 …>…</h3> */
  TRIM(
    SUBSTRING_INDEX(
      SUBSTRING_INDEX(s.content, '</h3>', 1),
      '>',
      -1
    )
  ) AS header,

  /* body: everything after first </h3> */
  TRIM(
    SUBSTRING(s.content, LOCATE('</h3>', s.content) + LENGTH('</h3>'))
  ) AS body

FROM wp_cop_snapshots s
JOIN (
  SELECT thread_id, MAX(version) AS max_version
  FROM wp_cop_snapshots
  GROUP BY thread_id
) mv
  ON mv.thread_id = s.thread_id
 AND mv.max_version = s.version

/* tie-break if multiple rows share the same (thread_id, version) */
JOIN (
  SELECT thread_id, version, MAX(id) AS max_id
  FROM wp_cop_snapshots
  GROUP BY thread_id, version
) mi
  ON mi.thread_id = s.thread_id
 AND mi.version   = s.version
 AND mi.max_id    = s.id

ORDER BY s.thread_id ASC;"
